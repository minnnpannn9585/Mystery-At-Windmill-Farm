def parse(path):
    data = Path(path).read_bytes()
    # Big-endian classic header, then version>=22 rewrites sizes as little-endian.
    meta, fsize, ver, off = struct.unpack_from(">IIII", data, 0)
    endian_flag = data[16]
    print(path)
    print("  classic", meta, fsize, ver, off, "endianflag", endian_flag)
    if ver >= 22:
        pos = 20
        metadata_size = struct.unpack_from("<I", data, pos)[0]
        pos += 4
        file_size = struct.unpack_from("<q", data, pos)[0]
        pos += 8
        data_offset = struct.unpack_from("<q", data, pos)[0]
        pos += 8
        unknown = struct.unpack_from("<q", data, pos)[0]
        pos += 8
        print("  v22", "meta", metadata_size, "fsize", file_size, "data", data_offset, "unk", unknown, "pos", pos)

        end = data.find(b"\x00", pos)

        return data, data_offset
    return data, off
